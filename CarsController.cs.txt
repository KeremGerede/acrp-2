using AnlıkAraç.Business;
using AnlıkAraç.Core.DTOs; // DTO'larımızı kullanabilmek için bu satırı ekledik
using Microsoft.AspNetCore.Http;
using Microsoft.AspNetCore.Mvc;

namespace AnlıkAraç.API.Controllers
{
    [Route("api/[controller]")]
    [ApiController]
    public class CarsController : ControllerBase
    {
        private readonly ICarService _carService;

        // Dependency Injection
        public CarsController(ICarService carService)
        {
            _carService = carService;
        }

        // GET: api/cars/available
        [HttpGet("available")]
        public async Task<IActionResult> GetAvailableCars()
        {
            // Artık veritabanı tablosu değil, vitrin modelimiz (CarListDto) listesi dönüyor
            var cars = await _carService.GetAllAvailableCarsAsync();
            return Ok(cars);
        }

        // GET: api/cars/5
        [HttpGet("{id}")]
        public async Task<IActionResult> GetCarById(int id)
        {
            var car = await _carService.GetCarByIdAsync(id);

            if (car == null)
            {
                return NotFound("Araç bulunamadı."); // HTTP 404
            }

            return Ok(car);
        }

        // POST: api/cars
        [HttpPost]
        public async Task<IActionResult> AddCar([FromBody] CarCreateDto carDto)
        {
            // DİKKAT: Artık dışarıdan 'Car' değil, güvenli 'CarCreateDto' nesnesini alıyoruz
            await _carService.AddCarAsync(carDto);
            return Ok("Araç sisteme başarıyla eklendi."); // HTTP 200
        }
    }
}
